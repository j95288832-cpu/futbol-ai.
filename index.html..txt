    <script>
        const apiKey = ""; // Gemini API key handled automatically
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

        document.getElementById('searchForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const club = document.getElementById('clubInput').value.trim();
            if (club) searchClub(club);
        });

        function searchClub(clubName) {
            document.getElementById('clubInput').value = clubName;
            fetchClubData(clubName);
        }

        async function fetchClubData(clubName) {
            const loadingState = document.getElementById('loadingState');
            const resultsContainer = document.getElementById('resultsContainer');
            const errorMessage = document.getElementById('errorMessage');
            const searchBtn = document.getElementById('searchBtn');

            // Toggle views
            loadingState.classList.remove('hidden');
            resultsContainer.classList.add('hidden');
            errorMessage.classList.add('hidden');
            searchBtn.disabled = true;

            const prompt = `Proporciona información estructurada sobre el club de fútbol "${clubName}". 
            Responde ÚNICAMENTE en formato JSON válido con el siguiente esquema estricto:

            {
              "name": "Nombre completo del club",
              "country": "País de origen",
              "badgeUrl": "URL directa de la imagen del escudo del club",
              "founded": "Año o fecha de fundación exacta",
              "founders": "Nombres de quiénes o cómo lo fundaron",
              "league": "Liga actual en la que compite (ej: Premier League, LaLiga, Liga Profesional, etc.)",
              "nickname": "Apodo o apodos del equipo",
              "stadium": {
                "name": "Nombre del estadio",
                "capacity": "Capacidad de espectadores"
              },
              "ranking": "Puesto aproximado en ranking global o histórico",
              "titles": {
                "continental": "Número exacto de Champions League o Copas Libertadores ganadas",
                "nationalLeague": "Número exacto de Ligas nacionales ganadas",
                "nationalCup": "Número de Copas nacionales ganadas",
                "breakdown": ["Lista de otros títulos importantes con cantidades"]
              },
              "topScorer": "Nombre del máximo goleador histórico y cantidad de goles",
              "history": "Un resumen conciso pero completo de su historia, origen y evolución.",
              "rivals": ["Lista de 2 a 4 sus máximos rivales tradicionales o clásicos"],
              "curiosities": ["3 a 4 datos curiosos, récords o hechos destacados del club"]
            }`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                tools: [{ "google_search": {} }],
                generationConfig: {
                    responseMimeType: "application/json"
                }
            };

            let retries = 3;
            let delay = 1000;

            while (retries > 0) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) throw new Error(`HTTP Error: ${response.status}`);

                    const result = await response.json();
                    const candidate = result.candidates?.[0];

                    if (candidate && candidate.content?.parts?.[0]?.text) {
                        const jsonText = candidate.content.parts[0].text;
                        const clubData = JSON.parse(jsonText);

                        const sources = [];
                        const groundingMetadata = candidate.groundingMetadata;
                        if (groundingMetadata && groundingMetadata.groundingAttributions) {
                            groundingMetadata.groundingAttributions.forEach(attr => {
                                if (attr.web?.uri && attr.web?.title) {
                                    sources.push({ uri: attr.web.uri, title: attr.web.title });
                                }
                            });
                        }

                        renderClubData(clubData, sources);
                        loadingState.classList.add('hidden');
                        resultsContainer.classList.remove('hidden');
                        searchBtn.disabled = false;
                        return;
                    } else {
                        throw new Error("Respuesta inválida de la IA.");
                    }

                } catch (err) {
                    console.warn(`Intento fallido. Reintentando...`, err);
                    retries--;
                    if (retries === 0) {
                        loadingState.classList.add('hidden');
                        showError("No pudimos obtener la información del club. Verifica el nombre e intenta nuevamente.");
                        searchBtn.disabled = false;
                        return;
                    }
                    await new Promise(res => setTimeout(res, delay));
                    delay *= 2;
                }
            }
        }

        function renderClubData(data, sources) {
            document.getElementById('clubName').innerText = data.name || "Club Desconocido";
            document.getElementById('countryBadge').innerText = data.country || "Internacional";
            document.getElementById('leagueName').innerText = data.league || "No disponible";
            document.getElementById('stadiumName').innerText = data.stadium?.name || "Estadio Principal";
            document.getElementById('stadiumCapacity').innerText = data.stadium?.capacity || "N/A";
            document.getElementById('foundedYear').innerText = data.founded || "Desconocido";
            document.getElementById('founders').innerText = data.founders || "Fundadores diversos";
            document.getElementById('founders').title = data.founders || "";
            document.getElementById('globalRanking').innerText = data.ranking || "Top";
            document.getElementById('nickname').innerText = data.nickname || "N/A";

            // Escudo
            const badgeImg = document.getElementById('clubBadge');
            const placeholderUrl = `https://placehold.co/120x120/1e293b/38bdf8?text=${encodeURIComponent(data.name ? data.name.substring(0,3).toUpperCase() : 'FC')}`;
            if (data.badgeUrl && data.badgeUrl.startsWith('http')) {
                badgeImg.src = data.badgeUrl;
            } else {
                badgeImg.src = placeholderUrl;
            }

            // Títulos
            document.getElementById('continentalTitles').innerText = data.titles?.continental || "0";
            document.getElementById('nationalLeagueTitles').innerText = data.titles?.nationalLeague || "0";
            document.getElementById('nationalCupTitles').innerText = data.titles?.nationalCup || "0";

            // Trofeos detallados
            const trophiesList = document.getElementById('trophiesList');
            trophiesList.innerHTML = "";
            if (data.titles?.breakdown && Array.isArray(data.titles.breakdown)) {
                data.titles.breakdown.forEach(item => {
                    const li = document.createElement('li');
                    li.className = "flex items-center gap-2 bg-slate-800/60 p-2.5 rounded-lg border border-slate-700/60";
                    li.innerHTML = `<i class="fa-solid fa-trophy text-amber-400 text-xs"></i> <span>${item}</span>`;
                    trophiesList.appendChild(li);
                });
            }

            // Historia
            document.getElementById('clubHistory').innerText = data.history || "Información histórica no disponible.";

            // Rivales
            const rivalsList = document.getElementById('rivalsList');
            rivalsList.innerHTML = "";
            if (data.rivals && Array.isArray(data.rivals)) {
                data.rivals.forEach(rival => {
                    const li = document.createElement('li');
                    li.className = "flex items-center gap-2 bg-slate-800/40 p-2 rounded-lg border border-slate-700/40";
                    li.innerHTML = `<i class="fa-solid fa-shield-halved text-red-400"></i> <span>${rival}</span>`;
                    rivalsList.appendChild(li);
                });
            }
            document.getElementById('topScorer').innerText = data.topScorer || "No registrado";

            // Curiosidades
            const factsList = document.getElementById('factsList');
            factsList.innerHTML = "";
            if (data.curiosities && Array.isArray(data.curiosities)) {
                data.curiosities.forEach(fact => {
                    const li = document.createElement('li');
                    li.className = "flex items-start gap-2.5 bg-slate-800/30 p-2.5 rounded-lg border border-slate-700/30";
                    li.innerHTML = `<i class="fa-solid fa-star text-emerald-400 text-xs mt-1 shrink-0"></i> <span>${fact}</span>`;
                    factsList.appendChild(li);
                });
            }

            // Fuentes
            const sourcesContainer = document.getElementById('sourcesContainer');
            const sourcesList = document.getElementById('sourcesList');
            sourcesList.innerHTML = "";
            if (sources && sources.length > 0) {
                sourcesContainer.classList.remove('hidden');
                sources.slice(0, 4).forEach(src => {
                    const a = document.createElement('a');
                    a.href = src.uri;
                    a.target = "_blank";
                    a.className = "inline-flex items-center gap-1.5 bg-slate-800 hover:bg-slate-700 px-3 py-1.5 rounded-lg text-blue-400 hover:text-blue-300 transition-all border border-slate-700";
                    a.innerHTML = `<i class="fa-solid fa-arrow-up-right-from-square text-[10px]"></i> <span class="truncate max-w-[160px]">${src.title}</span>`;
                    sourcesList.appendChild(a);
                });
            } else {
                sourcesContainer.classList.add('hidden');
            }
        }

        function showError(msg) {
            const errorMessage = document.getElementById('errorMessage');
            document.getElementById('errorText').innerText = msg;
            errorMessage.classList.remove('hidden');
        }
